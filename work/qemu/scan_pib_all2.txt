set pagination off
set confirm off
target remote :1234
set architecture aarch64
echo ===GDB-CONNECTED===
set $init = 0xffffffc00aaec240
set $node0 = $init + 0x4c8
set $cur = $node0
set $i = 0
while $i < 300
  set $t = $cur - 0x4c8
  set $pid = *(int*)($t + 0x5c8)
  set $pib = *(unsigned long*)($t + 0x898)
  if $pid > 100
    printf "pid=%d pib=%016lx\n", $pid, $pib
  end
  set $cur = *(unsigned long*)$cur
  set $i = $i + 1
  if $cur == $node0
    set $i = 9999
  end
end
printf "SCAN-END iter=%d\n", $i
detach
quit
