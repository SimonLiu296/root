set pagination off
set confirm off
target remote :1234
set architecture aarch64
echo ===GDB-CONNECTED===
set $init = 0xffffffc00aaec240
set $node0 = $init + 0x4c8
set $cur = $node0
set $i = 0
while $i < 300
  set $t = $cur - 0x4c8
  set $pib = *(unsigned long*)($t + 0x898)
  if $pib != 0
    printf "FOUND-DANGLING task=%016lx pid=%d pib=%016lx\n", $t, *(int*)($t+0x5c8), $pib
    printf "PIB_VALUE=%016lx\n", $pib
    echo ---dump-around-pib---\n
    x/60gx $pib-0x100
    echo ---end-dump---\n
  end
  set $cur = *(unsigned long*)$cur
  set $i = $i + 1
  if $cur == $node0
    set $i = 300
  end
end
printf "SCAN-END\n"
detach
quit
