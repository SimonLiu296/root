set pagination off
set confirm off
target remote :1234
set architecture aarch64
echo ===GDB-CONNECTED===
set $init = 0xffffffc00aaec240
set $node0 = $init + 0x4c8
set $cur = $node0
set $i = 0
while $i < 200
  set $t = $cur - 0x4c8
  set $pid = *(int*)($t + 0x5c8)
  set $pib = *(unsigned long*)($t + 0x898)
  if $pid > 100
    printf "LEADER pid=%d pib=%016lx\n", $pid, $pib
  end
  set $sig = *(unsigned long*)($t + 0x5d8)
  if $sig != 0
    set $thead = $sig + 0x5e8
    set $tnode = *(unsigned long*)$thead
    set $j = 0
    while $j < 32
      if $tnode == $thead
        set $j = 32
      end
      if $j < 32
        set $tt = $tnode - 0x678
        set $tpid = *(int*)($tt + 0x5c8)
        set $tpib = *(unsigned long*)($tt + 0x898)
        printf "THREAD pid=%d pib=%016lx\n", $tpid, $tpib
        set $tnode = *(unsigned long*)$tnode
        set $j = $j + 1
      end
    end
  end
  set $cur = *(unsigned long*)$cur
  set $i = $i + 1
  if $cur == $node0
    set $i = 9999
  end
end
printf "SCAN-END iter=%d\n", $i
detach
quit
