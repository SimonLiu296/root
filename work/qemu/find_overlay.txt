set pagination off
set confirm off
target remote :1234
set architecture aarch64
echo ===CONNECTED===
set $init = 0xffffffc00aaec240
set $node0 = $init + 0x4c8
set $cur = $node0
set $i = 0
while $i < 400
  set $t = $cur - 0x4c8
  set $pib = *(unsigned long*)($t + 0x898)
  if $pib != 0
    printf "DANGLING task=%016lx pid=%d pib=%016lx\n", $t, *(int*)($t+0x5c8), $pib
    printf "STACK-DUMP-AROUND-PIB:\n"
    x/40gx $pib-0x100
    echo ---END-DUMP---\n
  end
  set $cur = *(unsigned long*)$cur
  set $i = $i + 1
  if $cur == $node0
    set $i = 9999
  end
end
echo SCAN-COMPLETE\n
detach
quit
