set pagination off
set confirm off
target remote :1234
set architecture aarch64
echo ===GDB-CONNECTED===
set $init = 0xffffffc00aaec240
set $node0 = $init + 0x4c8
set $cur = $node0
set $i = 0
while $i < 300
  set $t = $cur - 0x4c8
  set $pib = *(unsigned long*)($t + 0x898)
  if $pib != 0
    printf "DANGLING task=%016lx pid=%d pib=%016lx\n", $t, *(int*)($t+0x5c8), $pib
    printf "PIB=%016lx\n", $pib
    set $stackbase = ($pib & 0xffffffffffffc000)
    printf "STACKBASE=%016lx\n", $stackbase
    echo ===FULLSTACK===
    x/2048gx $stackbase
    echo ===ENDFULLSTACK===
    printf "PIB_OFFSET_IN_STACK=%016lx\n", ($pib - $stackbase)
  end
  set $cur = *(unsigned long*)$cur
  set $i = $i + 1
  if $cur == $node0
    set $i = 300
  end
end
printf "SCAN-END\n"
detach
quit
