set pagination off
set confirm off
target remote :1234
set architecture aarch64
echo ===GDB-CONNECTED===
echo ---SEARCH-FPSIMD-MAGIC-IN-KERNEL-STACK---
printf "Looking for 0x46508001 (FPSIMD_MAGIC) in kernel stacks\n"
echo ---SCAN-TASK-LIST-FOR-WAITER---
set $init = 0xffffffc00aaec240
set $node0 = $init + 0x4c8
set $cur = $node0
set $i = 0
while $i < 200
  set $t = $cur - 0x4c8
  set $pid = *(int*)($t + 0x5c8)
  set $pib = *(unsigned long*)($t + 0x898)
  if $pid > 100
    printf "pid=%d pib=%016lx comm=", $pid, $pib
      x/s $t+0x790
    printf "  task=%016lx\n", $t
  end
  set $cur = *(unsigned long*)$cur
  set $i = $i + 1
  if $cur == $node0
    set $i = 9999
  end
end
echo ---SCAN-DONE---
detach
quit
